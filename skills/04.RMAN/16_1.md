```sql
[1] 시나리오 개요
-- 테이블이 잘못된 작업(TRUNCATE)으로 삭제된 상황을 가정한다.
-- TRUNCATE는 DDL 명령어로 UNDO 세그먼트를 남기지 않아 Flashback 기능을 사용할 수 없다.
-- RMAN의 RECOVER TABLE 기능을 사용하여 특정 시점의 테이블 단위 복구를 수행한다.
-- REMAP TABLE 옵션을 활용해 복구된 테이블을 임시 이름으로 적재하여 검증한 뒤 원본에 반영한다.
-- 복구 유형 : 테이블 단위 복구, 무중단(다른 업무에 영향 없음)

[2] 사전 조건
-- 이전 실습을 통해 생성된 전체 데이터베이스 백업 및 아카이브 로그가 존재해야 한다.
-- 보조 인스턴스(Auxiliary Instance)가 데이터를 복원할 임시 디스크 공간 확보가 필요하다.

[oracle@oel7v9 ~]$ mkdir -p /fra/aux
[oracle@oel7v9 ~]$ df -h /fra | tail -1
-- 보조 인스턴스가 사용할 임시 디렉터리 생성 및 여유 공간을 확인한다.

[oracle@oel7v9 ~]$ sqlplus -s / as sysdba <<< \

SELECT tablespace_name, ROUND(SUM(bytes)/1024/1024) AS mb
FROM dba_data_files
WHERE tablespace_name IN ('SYSTEM','SYSAUX','UNDOTBS1','USERS',)
GROUP BY tablespace_name ORDER BY 1;
-- 보조 인스턴스 기동 시 복원 대상이 되는 핵심 테이블스페이스(SYSTEM, SYSAUX, UNDO)와 타겟 테이블스페이스(USERS)의 총 요구 용량을 사전에 산정하여 디스크 공간을 검증한다.

[3] 초기 상태 확인
[oracle@oel7v9 ~]$ sqlplus / as sysdba
SYS@orcl> CREATE TABLE hr.emp160 TABLESPACE users AS SELECT * FROM hr.employees;
SYS@orcl> SELECT COUNT(*) FROM hr.emp160;
SYS@orcl> SELECT TO_CHAR(SYSDATE,'YYYY-MM-DD HH24:MI:SS') AS now FROM dual;
-- 복구 대상 테이블(hr.emp160)을 생성 및 데이터 건수 확인 후, 정상 상태의 현재 시점을 정확히 기록(예: 09:10:00)한다. 이 시점은 PITR(Point-In-Time Recovery)의 기준점이 된다.

SYS@orcl> EXIT

[oracle@oel7v9 ~]$ rman target / catalog rcatowner@rcat
RMAN> BACKUP DATABASE TAG 'BASE_161' PLUS ARCHIVELOG;
RMAN> EXIT
-- 실습 기준점 확립을 위해 데이터베이스와 아카이브 로그 전체 백업을 수행한다.

[4] 장애 유발 : 테이블 TRUNCATE
[oracle@oel7v9 ~]$ sqlplus / as sysdba
SYS@orcl> TRUNCATE TABLE hr.emp160;
SYS@orcl> SELECT COUNT(*) FROM hr.emp160;
SYS@orcl> ALTER SYSTEM ARCHIVE LOG CURRENT;
-- 사용자 실수로 인한 데이터 유실 상황을 시뮬레이션하기 위해 TRUNCATE 명령을 실행하고 데이터 건수가 0건이 되었음을 확인한다.

[5] 증상 관찰
SYS@orcl> SELECT COUNT(*) FROM hr.emp160
2  AS OF TIMESTAMP TO_TIMESTAMP('2025-05-18 09:10:00','YYYY-MM-DD HH24:MI:SS');
-- 장애 이전 시점으로 Flashback Query를 시도한다.
-- ORA-01466 에러가 발생한다. TRUNCATE로 인해 테이블 구조가 변경(데이터 블록 초기화)되었으며 UNDO 데이터를 남기지 않으므로 논리적 복구(Flashback)가 불가능함을 확인한다.

[6] 진단
SYS@orcl> EXIT
[oracle@oel7v9 ~]$ rman target / catalog rcatowner@rcat
RMAN> LIST BACKUP OF TABLESPACE system SUMMARY;
RMAN> LIST BACKUP OF TABLESPACE users SUMMARY;
-- 테이블 복구 시 보조 인스턴스가 필요로 하는 SYSTEM, SYSAUX, UNDO 및 USERS 테이블스페이스의 백업본이 리포지터리에 유효하게 존재하는지 점검한다.

[7] 복구 절차 : 오류 발생 → 원인 파악 → 수정
-- 테스트 1: 원본 이름 그대로 복구 시도
RMAN> RECOVER TABLE hr.emp160
2>   UNTIL TIME "TO_DATE('2025-05-18 09:10:00','YYYY-MM-DD HH24:MI:SS')"
3>   AUXILIARY DESTINATION '/fra/aux';
-- RMAN-05112(테이블이 이미 존재함) 에러가 발생한다. 원본 DB에 테이블 구조(껍데기)가 남아있으므로 덮어쓰기가 거부된다.

-- 테스트 2: AUXILIARY DESTINATION 생략
RMAN> RECOVER TABLE hr.emp160
2>   UNTIL TIME "TO_DATE('2025-05-18 09:10:00','YYYY-MM-DD HH24:MI:SS')"
3>   REMAP TABLE hr.emp160:emp160_recovered;
-- RMAN-05001/05517(파일 충돌) 에러가 발생한다. 보조 인스턴스가 파일 복원 시 사용할 임시 경로를 지정하지 않아 원본 운영 파일과 경로 충돌이 발생한다.

-- 정상 복구 절차 수행
RMAN> RECOVER TABLE hr.emp160
2>   UNTIL TIME "TO_DATE('2025-05-18 09:10:00','YYYY-MM-DD HH24:MI:SS')"
3>   AUXILIARY DESTINATION '/fra/aux'
4>   REMAP TABLE hr.emp160:emp160_recovered;

-- 수행 과정 요약:
-- 1. 지정된 경로(/fra/aux)에 보조 인스턴스(Auxiliary Instance) 자동 생성 및 기동
-- 2. 타겟 DB 컨트롤파일 복제(clone) 및 마운트
-- 3. 보조 인스턴스에 SYSTEM, SYSAUX, UNDOTBS1 테이블스페이스 복원 및 복구
-- 4. 대상 테이블스페이스(USERS) 복원 및 RESETLOGS 오픈
-- 5. Data Pump(EXPDP)를 호출하여 보조 인스턴스에서 테이블 덤프 파일 생성
-- 6. 원본 DB에서 Data Pump(IMPDP)를 호출하여 REMAP된 이름으로 테이블 임포트
-- 7. 보조 인스턴스 종료 및 사용된 임시 파일 일괄 정리

[8] 검증 및 실무 포인트
RMAN> EXIT
[oracle@oel7v9 ~]$ sqlplus / as sysdba

SYS@orcl> SELECT COUNT(*) FROM hr.emp160_recovered;
-- 복구된 테이블(emp160_recovered)을 조회하여 TRUNCATE 이전 시점의 데이터가 모두 복구되었음을 확인한다.

SYS@orcl> SELECT COUNT(*) FROM hr.emp160;
-- 원본 테이블은 0건 상태를 유지하고 있음을 확인한다(REMAP 옵션 적용 결과).

SYS@orcl> INSERT INTO hr.emp160 SELECT * FROM hr.emp160_recovered;
SYS@orcl> COMMIT;
SYS@orcl> DROP TABLE hr.emp160_recovered PURGE;
-- 데이터 정합성 검증이 완료되었으므로, 원본 테이블에 데이터를 적재하고 임시 테이블은 삭제한다.

SYS@orcl> EXIT
[oracle@oel7v9 ~]$ du -sh /fra/aux
-- 보조 인스턴스가 사용한 디렉터리의 용량이 0이 되어 임시 파일이 정상적으로 정리되었음을 확인한다.

[oracle@oel7v9 ~]$ ps -ef | grep [p]mon
-- 보조 인스턴스 관련 백그라운드 프로세스가 완전히 종료되었는지 확인한다.

-- 실무 포인트
-- 1. TRUNCATE, DROP TABLE ... PURGE 등 UNDO 데이터를 남기지 않는 DDL 작업으로 인한 데이터 유실 시 Flashback을 사용할 수 없으므로 RMAN의 RECOVER TABLE 기능이 필수적이다.
-- 2. RECOVER TABLE은 백업과 아카이브 로그를 기반으로 복구를 수행하며, 복구 과정 중 원본 데이터베이스의 다른 세션 작업에는 영향을 주지 않는다(무중단).
-- 3. 원본 테이블이 존재하는 상태에서 복구를 진행할 때는 반드시 REMAP TABLE 절을 사용하여 파일명 충돌을 방지하고 복구 데이터 검증 절차를 거친다.
-- 4. Fast Recovery Area(FRA)가 구성되지 않은 환경에서는 AUXILIARY DESTINATION 절을 통해 보조 인스턴스가 사용할 임시 디스크 경로를 명시적으로 할당해야 한다. 공간은 시스템 테이블스페이스 그룹과 대상 테이블스페이스를 모두 수용할 수 있어야 한다.
-- 5. 작업 완료 후 임시 디렉터리 내 파일 삭제 여부를 확인하여 추후 디스크 공간 부족(Full) 장애를 예방한다.
-- 6. RMAN-06962 / ORA-25153 오류 발생 시: Data Pump 작업 시 원본 DB의 임시 테이블스페이스(TEMP)가 비어있을 경우 발생한다. 타겟 DB에 TEMPFILE을 추가(ALTER TABLESPACE temp ADD TEMPFILE... REUSE)한 후 작업을 재시도하여 해결한다.
```
