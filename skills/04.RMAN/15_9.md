```sql
-- 잘못된 배치 작업으로 데이터가 지워졌다.
-- 그 작업 직전 시점으로 되돌린다.
-- 복원 지점을 만들어 두고 SET UNTIL RESTORE POINT 로 지정한다.
-- SET UNTIL의 순서 규칙과 오픈 전 검증 절차를 확인한다.
-- 복구 유형 : 불완전 복구, 전면 중단, 지정 시점 이후 손실

사전 조건
[세션: RMAN (Target + Catalog 동시 접속)]
BACKUP DATABASE TAG 'BASE_159' PLUS ARCHIVELOG;
깨끗한 복구 기준점(베이스 백업)을 마련합니다.

초기 상태 확인
[세션: SQL*Plus (SYSDBA)]
CREATE TABLE hr.orders159(id NUMBER, memo VARCHAR2(30)) TABLESPACE users;
INSERT INTO hr.orders159 SELECT LEVEL, 'order-' || LEVEL FROM dual CONNECT BY LEVEL <= 1000;
COMMIT;
정상적인 1,000건의 샘플 테이블 데이터를 적재합니다.

SELECT COUNT(*) FROM hr.orders159;

ALTER SYSTEM ARCHIVE LOG CURRENT;

CREATE RESTORE POINT before_batch_159;
위험한 대규모 배치(Batch) 작업 직전에, 최후의 보루로 손쉽게 돌아갈 수 있도록 복원 지점(Restore Point)을 생성해 둡니다.

SELECT name, scn, TO_CHAR(time, 'YYYY-MM-DD HH24:MI:SS') AS t 
FROM v$restore_point;
NAME                 SCN       T  
----------         --------    -----------

BEFORE_BATCH_159   4902396    2026-09-21 11:50:18


생성된 복원 지점의 이름, SCN 번호, 타임스탬프를 딕셔너리에서 확인합니다.

장애 유발 : 잘못된 배치 작업
[세션: SQL*Plus (SYSDBA)]
DELETE FROM hr.orders159 WHERE id > 100;
COMMIT;
개발자나 운영자의 실수로 WHERE 조건을 잘못 주어 애먼 데이터 900건이 날아가는 대참사(논리적 오류)를 시뮬레이션합니다.

SELECT COUNT(*) FROM hr.orders159;
100건만 남은 것을 확인합니다.

INSERT INTO hr.orders159 VALUES (9001, 'after batch');
COMMIT;
장애(삭제)가 발생한 사실을 인지하지 못한 채, 이후에도 정상적인 업무가 계속 진행되어 새로운 트랜잭션이 쌓인 상황을 만듭니다.

ALTER SYSTEM ARCHIVE LOG CURRENT;

[세션: RMAN (Target + Catalog 동시 접속)]
BACKUP DATABASE TAG 'NIGHTLY_159' PLUS ARCHIVELOG;
데이터가 꼬인 상태 그대로 야간 정기 백업(Nightly Backup)이 돌아가 버린 최악의 시나리오를 가정합니다.

증상 관찰
[세션: SQL*Plus (SYSDBA)]
SELECT COUNT(*) FROM hr.orders159;

SELECT * FROM v$recover_file;
물리적인 디스크나 파일 손상이 아니라 테이블 내부 데이터만 꼬인 '논리적 장애'이므로, 오라클 백그라운드 프로세스는 이를 문제로 인지하지 않아 복구 요구 파일 목록(v$recover_file)은 깨끗하게 비어있습니다. 결국 데이터베이스 전체를 과거 특정 시점(Point-In-Time)으로 통째로 타임머신을 태워야 합니다.

진단
[세션: RMAN (Target + Catalog 동시 접속)]
RESTORE DATABASE UNTIL RESTORE POINT before_batch_159 PREVIEW SUMMARY;
과거 시점으로 시점 기반 불완전 복구(PITR)를 할 때, 어느 백업 세트와 아카이브가 사용되는지 미리 조회(PREVIEW)해 봅니다. UNTIL 절을 주었기 때문에 가장 최신 백업(NIGHTLY_159)은 배제되고 장애 발생 전의 안전한 옛날 백업(BASE_159)이 영리하게 선택됨을 확인합니다.

RESTORE DATABASE PREVIEW SUMMARY;
반면, UNTIL 제약 없이 무작정 조회를 돌리면 당연히 꼬인 데이터가 포함된 가장 최신 백업(NIGHTLY_159)이 잡히게 됩니다.

복구 절차 : 오류 발생 → 원인 파악 → 수정
[세션: RMAN (Target + Catalog 동시 접속)]
SHUTDOWN IMMEDIATE
STARTUP MOUNT
전체 데이터베이스 불완전 복구를 위해 인스턴스를 내리고 MOUNT 상태로 올립니다.
RESTORE DATABASE;
RUN {
   SET UNTIL RESTORE POINT before_batch_159;
   RECOVER DATABASE;
}

[순서 오류 테스트] 무작정 RESTORE를 먼저 실행하고 뒤늦게 RUN 블록 안에서 SET UNTIL을 지정해 복구해 봅니다.
RESTORE는 시점 제약이 없었으므로 최신 야간 백업(NIGHTLY_159)을 복원해 버렸습니다. 그런데 직후 RECOVER는 과거 시점(before_batch_159)까지만 밀어 넣으려 하니, "이미 미래 시점(야간 백업)의 파일이 복원되어 있어서 과거로 돌아갈 수 없어!"라며 RMAN-06556 에러를 뱉고 실패합니다.

RUN {
   SET UNTIL RESTORE POINT before_batch_159;
   RESTORE DATABASE;
   RECOVER DATABASE;
}

[올바른 순서] SET UNTIL 구문은 RECOVER뿐만 아니라 RESTORE가 올바른(과거의) 백업 피스를 고르는 기준점 역할도 하므로 반드시 RUN 블록의 맨 첫 줄에 배치해야 합니다.
SET UNTIL: 타겟 시점을 확정합니다.
RESTORE: 타겟 시점보다 앞선 안전한 과거 백업(BASE_159)을 덮어씁니다.
RECOVER: 과거 백업에 아카이브를 부어 타겟 시점(배치 직전) 직전까지만 딱 맞게 롤포워드(Roll-forward)합니다.
```
